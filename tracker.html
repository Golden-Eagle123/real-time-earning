        <!-- (1) THE RATE INPUT BAR -->

        <div class="rate-bar">

            <div class="rate-bar-input-group">

                <span>$</span>

                <span>/hr</span>

            </div>

        </div>


        <!-- (2) THE MAIN ACTION CARD -->

        <div class="card">

            <span id="statusBadge" class="status-badge">Clocked Out</span>

            

            <!-- This is the live session ticker -->

            <h1 id="sessionTicker" class="ticker session-only">$0.00</h1>

            <p class="sub-ticker session-only">Current Session</p>


            <button id="startBtn" class="btn btn-start stopped-only" onclick="toggleClock()">Clock In & Start Tracking</button>

            <button id="stopBtn" class="btn btn-stop session-only" onclick="toggleClock()">Clock Out & Save</button>

        </div>


        <!-- (3) CUMULATIVE DASHBOARD -->

        <div class="dashboard-grid">

            <div class="stat-box">

                <div class="stat-label">Total Earned</div>

                <div id="cumulativeTotal" class="stat-value">$0.00</div>

            </div>

            <div class="stat-box">

                <div class="stat-label">Last Session</div>

                <div id="lastSessionValue" class="stat-value cumulative">$0.00</div>

            </div>

        </div>


    </div>


    <script>

        // --- (A) Core App Variables (State) ---

        let startTime, timerInterval;

        let rate = 50;

        let cumulativeTotal = 0; // Everything earned *before* this session

        let isWorking = false;


        // --- (B) Initial Load: Retrieve Saved Data (The Memory) ---

        window.onload = function() {

            // Restore Hourly Rate

            const savedRate = localStorage.getItem('earningsRate');

            if (savedRate) {

                rate = parseFloat(savedRate);

                document.getElementById('rateInput').value = rate;

            }


            // Restore Cumulative Total

            const savedCumulative = localStorage.getItem('earningsCumulativeTotal');

            if (savedCumulative) {

                cumulativeTotal = parseFloat(savedCumulative);

            }


            // Restore last session info for the dashboard

            const savedLastSession = localStorage.getItem('earningsLastSessionEarned');

            if(savedLastSession) {

                document.getElementById('lastSessionValue').innerText = "$" + parseFloat(savedLastSession).toFixed(2);

            }


            // UPDATE DASHBOARD with cumulative data

            document.getElementById('cumulativeTotal').innerText = "$" + cumulativeTotal.toFixed(2);

            document.getElementById('sessionTicker').innerText = "$0.00"; // Reset session display

        };



        // --- (C) Core Functions ---


        function updateRate() {

            rate = parseFloat(document.getElementById('rateInput').value);

            localStorage.setItem('earningsRate', rate); // SAVE to memory immediately

        }


        function toggleClock() {

            isWorking = !isWorking;

            const body = document.getElementById('mainBody');

            const statusBadge = document.getElementById('statusBadge');


            if (isWorking) {

                // ACTIONS UPON CLOCKING IN

                startTime = Date.now();

                body.classList.add('working-active'); // Changes UI view

                statusBadge.innerText = "WORKING LIVE";

                statusBadge.classList.add('working');


                // Starts the fast ticker update loop

                timerInterval = setInterval(updateTicker, 50); // Updates every 50ms for smoother animation

            } else {

                // ACTIONS UPON CLOCKING OUT

                clearInterval(timerInterval);

                body.classList.remove('working-active');

                statusBadge.innerText = "Clocked Out";

                statusBadge.classList.remove('working');


                saveAndResetSession();

            }

        }


        // --- (D) The Math & Memory Functions ---


        function getSessionEarnings() {

            // Calculate how much was earned *only* in this current clock-in session

            if (!startTime) return 0;

            const elapsedSecs = (Date.now() - startTime) / 1000;

            return (rate / 3600) * elapsedSecs;

        }


        function updateTicker() {

            // This runs repeatedly during a work session

            const currentSessionEarned = getSessionEarnings();

            

            // 1. Update the 'Current Session' live display

            // We use 2 decimals for cleaner view in the session card, but 4 would work too.

            document.getElementById('sessionTicker').innerText = "$" + currentSessionEarned.toFixed(2);


            // 2. We can show a 'Projected' cumulative total on the dashboard in real-time.

            // This adds the *previous cumulative* plus the *live session*

            const projectedTotal = cumulativeTotal + currentSessionEarned;

            document.getElementById('cumulativeTotal').innerText = "$" + projectedTotal.toFixed(2);

        }


        function saveAndResetSession() {

            // When Clocking Out, take the final session earnings and finalize the memory.

            const sessionEarned = getSessionEarnings();


            // 1. Add session earnings to the persistent cumulative total

            cumulativeTotal += sessionEarned;


            // 2. SAVE the new finalized data to memory (localStorage)

            localStorage.setItem('earningsCumulativeTotal', cumulativeTotal);

            localStorage.setItem('earningsLastSessionEarned', sessionEarned);


            // 3. Reset the dashboard UI elements for the stopped state

            document.getElementById('cumulativeTotal').innerText = "$" + cumulativeTotal.toFixed(2);

            document.getElementById('lastSessionValue').innerText = "$" + sessionEarned.toFixed(2);

            document.getElementById('sessionTicker').innerText = "$0.00";

            

            // Reset state variables

            startTime = null; 

        }


    </script>
